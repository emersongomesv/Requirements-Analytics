-- Script 02: Criação das Tabelas (DDL)

CREATE TABLE PROJETO (
    ID_Projeto INT IDENTITY(1,1) PRIMARY KEY,
    Nome_Projeto VARCHAR(100) NOT NULL,
    Responsavel VARCHAR(100) NOT NULL,
    Data_Inicio DATE NOT NULL,
    Data_Fim DATE NULL
);

CREATE TABLE USUARIO (
    ID_Usuario INT IDENTITY(1,1) PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    Perfil VARCHAR(50) NOT NULL,
    Area VARCHAR(50) NOT NULL
);

CREATE TABLE STAKEHOLDER (
    ID_Stakeholder INT IDENTITY(1,1) PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    Area VARCHAR(50) NOT NULL,
    Cargo VARCHAR(50) NOT NULL
);

CREATE TABLE REQUISITO (
    ID_Requisito INT IDENTITY(1,1) PRIMARY KEY,
    ID_Projeto INT NOT NULL,
    Titulo VARCHAR(200) NOT NULL,
    Descricao VARCHAR(MAX) NULL,
    Prioridade VARCHAR(20) NOT NULL,
    Status VARCHAR(50) NOT NULL,
    Data_Criacao DATE NOT NULL,
    Data_Entrada_Analise DATE NULL,
    Data_Aprovacao DATE NULL,
    CONSTRAINT FK_REQUISITO_PROJETO FOREIGN KEY (ID_Projeto) REFERENCES PROJETO(ID_Projeto)
);

CREATE TABLE APROVACAO (
    ID_Aprovacao INT IDENTITY(1,1) PRIMARY KEY,
    ID_Requisito INT NOT NULL,
    ID_Stakeholder INT NOT NULL,
    Status_Aprovacao VARCHAR(50) NOT NULL,
    Data_Aprovacao DATE NULL,
    CONSTRAINT FK_APROVACAO_REQUISITO FOREIGN KEY (ID_Requisito) REFERENCES REQUISITO(ID_Requisito),
    CONSTRAINT FK_APROVACAO_STAKEHOLDER FOREIGN KEY (ID_Stakeholder) REFERENCES STAKEHOLDER(ID_Stakeholder)
);

CREATE TABLE HISTORICO_REQUISITO (
    ID_Historico INT IDENTITY(1,1) PRIMARY KEY,
    ID_Requisito INT NOT NULL,
    Campo_Alterado VARCHAR(100) NOT NULL,
    Valor_Anterior VARCHAR(MAX) NULL,
    Novo_Valor VARCHAR(MAX) NULL,
    ID_Usuario INT NOT NULL,
    Data_Alteracao DATETIME NOT NULL,
    CONSTRAINT FK_HISTORICO_REQUISITO FOREIGN KEY (ID_Requisito) REFERENCES REQUISITO(ID_Requisito),
    CONSTRAINT FK_HISTORICO_USUARIO FOREIGN KEY (ID_Usuario) REFERENCES USUARIO(ID_Usuario)
);
